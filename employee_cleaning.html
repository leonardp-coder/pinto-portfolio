Normalize Text Columns & Whitespace

clean_text_fields.py Python · pandas 
import pandas as pd

text_cols = ["full_name", "department", "job_role", "location"]
for col in text_cols:
    if col in df.columns:
        df[col] = (
            df[col]
            .astype("string")
            .str.strip()
            .str.replace(r"\s+", " ", regex=True)
        )

if "department" in df.columns:
    df["department"] = df["department"].str.title()
